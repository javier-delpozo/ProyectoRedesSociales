45    def importar(self, ruta: str | Path, nombre: str | None = None) -> Conversacion:
46        """Sobrescribe el contrato de :class:`Importador` para el formato CSV."""
47        origen = self.validar_ruta(ruta)
48        conversacion = Conversacion(nombre or origen.stem)
49        try:
50            with origen.open("r", encoding="utf-8-sig", newline="") as fichero:
51                lector = csv.DictReader(fichero)
52                columnas = set(lector.fieldnames or [])
53                faltantes = self.COLUMNAS - columnas
54                if faltantes:
55                    detalle = ", ".join(sorted(faltantes))
56                    raise ErrorImportacion(f"Faltan columnas obligatorias: {detalle}")
57                for numero_linea, fila in enumerate(lector, start=2):
58                    conversacion.agregar(self._crear_mensaje(fila, numero_linea))
59        except UnicodeDecodeError as exc:
60            raise ErrorImportacion("El CSV debe estar codificado en UTF-8") from exc
61        except OSError as exc:
62            raise ErrorImportacion(f"No se pudo leer {origen}: {exc}") from exc
63
64        if not conversacion.mensajes:
65            raise ErrorImportacion("El CSV no contiene mensajes")
66        logger.info("Importados %d mensajes desde %s", len(conversacion.mensajes), origen)
67        return conversacion

69    @staticmethod
70    def _crear_mensaje(fila: dict[str, str | None], numero_linea: int) -> Mensaje:
71        try:
72            autor = (fila.get("autor") or "").strip()
73            contenido = (fila.get("contenido") or "").strip()
74            fecha_texto = (fila.get("fecha") or "").strip()
75            fecha = datetime.fromisoformat(fecha_texto.replace("Z", "+00:00"))

77        except (ValueError, ErrorValidacion) as exc:
78            raise ErrorImportacion(f"Fila {numero_linea} inválida: {exc}") from exc

 41    def __init__(self, ruta: str | Path) -> None:
 42        self.ruta = Path(ruta)

 44    def inicializar(self) -> None:
 45        try:
 46            self.ruta.parent.mkdir(parents=True, exist_ok=True)
 47            with self._conexion() as conexion:
 48                conexion.executescript(ESQUEMA)
 49        except (OSError, sqlite3.Error) as exc:
 50            raise ErrorPersistencia(f"No se pudo inicializar la base de datos: {exc}") from exc
 51
 52    def guardar(self, conversacion: Conversacion) -> int:
 53        self.inicializar()
 54        try:
 55            with self._conexion() as conexion:
 56                cursor = conexion.execute(
 57                    "INSERT INTO conversaciones(nombre) VALUES (?)", (conversacion.nombre,)
 58                )
 59                conversacion_id = int(cursor.lastrowid)
 60                conexion.executemany(
 61                    """INSERT INTO mensajes(conversacion_id, autor, contenido, fecha)
 62                       VALUES (?, ?, ?, ?)""",
 63                    [
 64                        (conversacion_id, m.autor.nombre, m.contenido, m.fecha.isoformat())
 65                        for m in conversacion.mensajes
 66                    ],
 67                )
 68        except sqlite3.Error as exc:
 69            raise ErrorPersistencia(f"No se pudo guardar la conversación: {exc}") from exc
 70
 71        conversacion.id = conversacion_id
 72        logger.info(
 73            "Guardada conversación %d (%d mensajes)",
 74            conversacion_id,
 75            len(conversacion.mensajes),
 76        )
 77        return conversacion_id
 78
 79    def obtener(self, conversacion_id: int) -> Conversacion:
 80        self.inicializar()
 81        try:
 82            with self._conexion() as conexion:
 83                fila = conexion.execute(
 84                    "SELECT id, nombre FROM conversaciones WHERE id = ?", (conversacion_id,)
 85                ).fetchone()
 86                if fila is None:
 87                    raise ConversacionNoEncontrada(
 88                        f"No existe la conversación con id {conversacion_id}"
 89                    )
 90                mensajes = conexion.execute(
 91                    """SELECT autor, contenido, fecha FROM mensajes
 92                       WHERE conversacion_id = ? ORDER BY fecha, id""",
 93                    (conversacion_id,),
 94                ).fetchall()
 95        except ConversacionNoEncontrada:
 96            raise
 97        except sqlite3.Error as exc:
 98            raise ErrorPersistencia(f"No se pudo leer la conversación: {exc}") from exc
 99
100        conversacion = Conversacion(fila["nombre"], id=fila["id"])
101        for mensaje in mensajes:
102            conversacion.agregar(
103                Mensaje(
104                    Usuario(mensaje["autor"]),
105                    mensaje["contenido"],
106                    datetime.fromisoformat(mensaje["fecha"]),
107                )
108            )
109        return conversacion

111    def listar(self) -> list[dict[str, object]]:
112        self.inicializar()
113        try:
114            with self._conexion() as conexion:
115                filas = conexion.execute(
116                    """SELECT c.id, c.nombre, c.creada_en, COUNT(m.id) AS mensajes
117                       FROM conversaciones c LEFT JOIN mensajes m ON m.conversacion_id = c.id
118                       GROUP BY c.id ORDER BY c.id"""
119                ).fetchall()
120        except sqlite3.Error as exc:
121            raise ErrorPersistencia(f"No se pudieron listar las conversaciones: {exc}") from exc
122        return [dict(fila) for fila in filas]

140    @contextmanager
141    def _conexion(self) -> Iterator[sqlite3.Connection]:

145            with conexion:
146                yield conexion
